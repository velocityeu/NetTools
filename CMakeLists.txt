cmake_minimum_required(VERSION 3.25)
project(VelocityNetTools VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build with MSVC for Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /permissive- /utf-8 /EHsc /Zc:__cplusplus)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_custom_target(native_resources
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/scripts/build_native_resources.py"
  WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" VERBATIM)
file(GLOB NET_SOURCES CONFIGURE_DEPENDS src/net/*.cpp)
add_library(veu_subnet STATIC src/core/subnet.cpp src/core/classification.cpp)
target_include_directories(veu_subnet PUBLIC include)
add_library(veu_network STATIC ${NET_SOURCES})
target_include_directories(veu_network PUBLIC include)
target_link_libraries(veu_network PUBLIC ws2_32 iphlpapi dnsapi winhttp crypt32)
add_library(veu_storage STATIC src/storage.cpp)
target_include_directories(veu_storage PUBLIC include)
target_link_libraries(veu_storage PUBLIC ole32)
add_executable(VelocityNetTools WIN32 src/main.cpp src/ui/subnet_pane.cpp src/ui/network_pane.cpp src/ui/help.cpp resources/veu_nettools.rc)
add_dependencies(VelocityNetTools native_resources)
target_include_directories(VelocityNetTools PRIVATE include resources)
target_link_libraries(VelocityNetTools PRIVATE veu_subnet veu_network veu_storage comctl32 uxtheme ole32 shell32 shlwapi comdlg32 gdi32 user32 windowscodecs)
target_link_options(VelocityNetTools PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf /DEBUG /PDBALTPATH:%_PDB%)
target_compile_options(VelocityNetTools PRIVATE /guard:cf)
enable_testing()
add_executable(subnet_tests tests/subnet_tests.cpp)
target_link_libraries(subnet_tests PRIVATE veu_subnet)
add_test(NAME subnet COMMAND subnet_tests)
add_executable(network_tests tests/network_tests.cpp)
target_link_libraries(network_tests PRIVATE veu_network)
add_test(NAME network COMMAND network_tests)
add_executable(storage_tests tests/storage_tests.cpp)
target_link_libraries(storage_tests PRIVATE veu_storage)
add_test(NAME storage COMMAND storage_tests)
add_test(NAME resources COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/resources_tests.py")
add_test(NAME help_fresh COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/scripts/build_help.py" --check)
add_test(NAME seo_fresh COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/scripts/build_seo.py" --check)

set(VEU_VERSION "0.1.0-dev" CACHE STRING "SemVer embedded in the executable")
set(VEU_BUILD_ID "local" CACHE STRING "Public build identifier; never a local path")
set(VEU_VERSION_BUILD "0" CACHE STRING "16-bit Windows file build number")
if(NOT VEU_VERSION MATCHES "^([0-9]+)\\.([0-9]+)\\.([0-9]+)(-[0-9A-Za-z.-]+)?(\\+[0-9A-Za-z.-]+)?$")
  message(FATAL_ERROR "VEU_VERSION must be a semantic version with optional build metadata.")
endif()
set(VEU_MAJOR "${CMAKE_MATCH_1}")
set(VEU_MINOR "${CMAKE_MATCH_2}")
set(VEU_PATCH "${CMAKE_MATCH_3}")
foreach(PART VEU_MAJOR VEU_MINOR VEU_PATCH VEU_VERSION_BUILD)
  if(NOT "${${PART}}" MATCHES "^[0-9]+$" OR ${${PART}} GREATER 65535)
    message(FATAL_ERROR "${PART} must be between 0 and 65535.")
  endif()
endforeach()
if(NOT VEU_BUILD_ID MATCHES "^[0-9A-Za-z._-]+$")
  message(FATAL_ERROR "VEU_BUILD_ID may only contain public identifier characters.")
endif()
target_compile_definitions(VelocityNetTools PRIVATE VEU_APP_VERSION_MAJOR=${VEU_MAJOR}
 VEU_APP_VERSION_MINOR=${VEU_MINOR} VEU_APP_VERSION_PATCH=${VEU_PATCH}
 VEU_APP_VERSION_BUILD=${VEU_VERSION_BUILD} VEU_APP_VERSION_STRING="${VEU_VERSION}"
 VEU_BUILD_ID="${VEU_BUILD_ID}")
foreach(NAME subnet network)
  if(EXISTS "${CMAKE_SOURCE_DIR}/tests/${NAME}_pane_tests.cpp")
    add_executable(${NAME}_pane_tests tests/${NAME}_pane_tests.cpp src/ui/${NAME}_pane.cpp)
    target_link_libraries(${NAME}_pane_tests PRIVATE veu_${NAME} veu_storage comctl32 comdlg32 uxtheme user32 gdi32 ole32 shell32)
    add_test(NAME ${NAME}_pane COMMAND ${NAME}_pane_tests)
  endif()
endforeach()
add_test(NAME network_loopback COMMAND network_tests --integration)

add_executable(frame_smoke tests/frame_smoke.cpp src/ui/subnet_pane.cpp src/ui/network_pane.cpp src/ui/help.cpp resources/veu_nettools.rc)
add_dependencies(frame_smoke native_resources)
target_include_directories(frame_smoke PRIVATE include resources)
target_link_libraries(frame_smoke PRIVATE veu_subnet veu_network veu_storage comctl32 uxtheme ole32 shell32 shlwapi comdlg32 gdi32 user32 windowscodecs)
add_test(NAME frame_smoke COMMAND frame_smoke)
set_tests_properties(frame_smoke network_loopback PROPERTIES TIMEOUT 60)

target_link_options(frame_smoke PRIVATE /MANIFEST:NO)

add_test(NAME workflow_metadata COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/workflow_metadata_tests.py")
